Parámetros#

tidy

Valores devueltos#

Ejemplos#

Ejemplo de tidy::head

<?php
$html = '
<html>
  <head>
    <title>test</title>
  </head>
  <body>
    <p>parrafo</p>
  </body>
</html>';

$tidy = tidy_parse_string($html);

$head = $tidy->head();
echo $head->value;
?>
